cmake_minimum_required(VERSION 3.22)
project(spool_textmate C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static regex engine" FORCE)
set(BUILD_TEST OFF CACHE BOOL "No upstream executables" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "No upstream executables" FORCE)
add_subdirectory("${ONIG_SOURCE}" oniguruma EXCLUDE_FROM_ALL)
add_library(spool-textmate SHARED syntax.c)
target_include_directories(spool-textmate PRIVATE "${ONIG_SOURCE}/src")
if(NOT ANDROID)
    if(EXISTS "${JAVA_HOME}/include/jni.h")
        target_include_directories(spool-textmate PRIVATE "${JAVA_HOME}/include" "${JAVA_HOME}/include/${JNI_PLATFORM}")
    else()
        # Studio's JBR can omit headers. NDK jni.h is VM-neutral; never shadow host libc headers.
        target_compile_options(spool-textmate PRIVATE -idirafter "${JNI_NDK_INCLUDE}")
    endif()
endif()
target_link_libraries(spool-textmate PRIVATE onig)
